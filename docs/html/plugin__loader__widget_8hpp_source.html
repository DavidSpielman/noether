62 template <typename PluginT>
63 PluginLoaderWidget<PluginT>::~PluginLoaderWidget()
64 {
65  /* By default Qt maintains ownership of all widgets and deletes them automatically in the destructor of the QWidget
66  * parent class (which occurs after the execution of the destructor of this class). When the plugins are deleted
67  * during the destruction of this class (before the QWidget parent class deletes the widgets created by them), the
68  * library in which the plugins (and the widgets they create) are defined will be unloaded. Then when the QWidget base
69  * class tries to invoke the destructor of the widget(s) created by the plugin(s), a segfault will occur because the
70  * library defining those destructor(s) will have already been unloaded.
71  *
72  * The solution to this issue is to delete all widgets loaded from plugin(s) before deleting the plugins and plugin
73  * loader.
74  */
75  while (ui_->stacked_widget->count() > 0)
76  {
77  QWidget* widget = ui_->stacked_widget->widget(0);
78  ui_->stacked_widget->removeWidget(widget);
79  delete widget;
80  }
81 }
